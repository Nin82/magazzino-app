function handleScaricoBarcodeInput() { const barcodeInput = document.getElementById('scarico-barcode'); const select = document.getElementById('scarico-select'); const giacenzaDisplay = document.getElementById('scarico-giacenza-display'); const barcodeValue = barcodeInput.value.trim(); // Reset della variabile di stato if (typeof currentScannedItem !== 'undefined') { currentScannedItem = null; } if (scaricoBarcodeTypingTimer) clearTimeout(scaricoBarcodeTypingTimer); giacenzaDisplay.textContent = 'Giacenza Attuale: -'; // Se vuoto, resetta i blocchi e il select if (barcodeValue.length === 0) { select.value = ''; barcodeInput.disabled = false; select.disabled = false; return; } // Debounce: attende il tempo necessario per una scansione completa (1000ms) scaricoBarcodeTypingTimer = setTimeout(() => { const foundItem = magazzinoData.find(item => String(item.barcode).trim() === barcodeValue ); if (foundItem) { // Articolo trovato con match esatto Barcode currentScannedItem = foundItem; // Seleziona l'articolo corretto nel dropdown select.value = foundItem.objectId; giacenzaDisplay.textContent = `Giacenza Attuale: ${foundItem.quantity || 0} unità`; // Blocca il barcode e il select per non confondere l'utente barcodeInput.disabled = true; select.disabled = true; //showNotification(`Articolo trovato: ${foundItem.article}`, 'info'); } else { // Articolo non trovato: resetta la selezione select.value = ''; barcodeInput.disabled = false; select.disabled = false; // showNotification('Barcode non trovato. Riprova o Seleziona manualmente.', 'warning'); } }, BARCODE_DEBOUNCE_DELAY); }
