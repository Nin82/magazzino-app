// ====================== QR SCANNER ROBUSTO ====================== window.videoStream = null; window.html5QrInstance = null; window.lastScanned = ''; window.scanCooldown = false; window.isScanning = false; async function startQRScanner() { if(window.isScanning) return; // previene doppi start window.isScanning = true; const video = document.getElementById('qr-video'); const resultInput = document.getElementById('qr-result'); const statusLabel = document.getElementById('qr-status'); try { if ('BarcodeDetector' in window) { window.videoStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } }); video.srcObject = window.videoStream; const detector = new BarcodeDetector({ formats: ['qr_code', 'code_128', 'ean_13', 'ean_8'] }); const scanLoop = async () => { if(!window.videoStream) return; try { const barcodes = await detector.detect(video); if(barcodes.length > 0 && barcodes[0].rawValue !== window.lastScanned && !window.scanCooldown){ window.lastScanned = barcodes[0].rawValue; resultInput.value = window.lastScanned; showNotification(`📷 Codice rilevato: ${window.lastScanned}`, 'success'); playBeep(150, 600, 0.5); window.scanCooldown = true; setTimeout(()=> window.scanCooldown=false, 1000); } } catch(err){ console.warn(err); } requestAnimationFrame(scanLoop); }; requestAnimationFrame(scanLoop); } else { if(!window.html5QrInstance) window.html5QrInstance = new Html5Qrcode("qr-video"); await window.html5QrInstance.start( { facingMode: "environment" }, { fps: 10, qrbox: Math.min(window.innerWidth*0.8, 300) }, qrCodeMessage => { if(qrCodeMessage !== window.lastScanned && !window.scanCooldown){ window.lastScanned = qrCodeMessage; resultInput.value = window.lastScanned; showNotification(`📷 Codice rilevato: ${window.lastScanned}`, 'success'); playBeep(150, 600, 0.5); window.scanCooldown = true; setTimeout(()=> window.scanCooldown=false, 1000); } }, errorMessage => { /* ignora errori */ } ); } if(statusLabel){ statusLabel.textContent = '📸 Scanner attivo'; statusLabel.style.color = '#16a34a'; } } catch(err){ console.error('Errore fotocamera:', err); alert('Impossibile accedere alla fotocamera. Controlla permessi e browser.'); window.isScanning = false; } } function stopQRScanner(){ const statusLabel = document.getElementById('qr-status'); const resultInput = document.getElementById('qr-result'); window.lastScanned = ''; window.scanCooldown = false; window.isScanning = false; if(resultInput) resultInput.value = ''; if(window.videoStream){ window.videoStream.getTracks().forEach(track => track.stop()); window.videoStream = null; } if(window.html5QrInstance){ window.html5QrInstance.stop() .then(()=>{ window.html5QrInstance.clear(); window.html5QrInstance = null; }) .catch(err => console.warn('Errore stop QR:', err)); } if(statusLabel){ statusLabel.textContent = '⛔ Scanner disattivato'; statusLabel.style.color = '#dc2626'; } } // Bottone Stop Scanner document.getElementById('stop-qr').addEventListener('click', stopQRScanner); // Switch tab start/stop document.querySelectorAll('.tab-container button').forEach(btn => { btn.addEventListener('click', async ()=>{ document.querySelectorAll('.tab-container button').forEach(b => b.classList.remove('active')); btn.classList.add('active'); document.querySelectorAll('[role="tabpanel"]').forEach(tab => tab.classList.add('hidden')); document.getElementById(btn.dataset.target).classList.remove('hidden'); if(btn.dataset.target === 'tab-qr') await startQRScanner(); else stopQRScanner(); }); });
